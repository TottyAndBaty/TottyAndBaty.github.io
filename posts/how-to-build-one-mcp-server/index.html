如何使用golang创建一个自己的MCP

2026年8月4日 · 1 分钟 · 164 字 · beyond

用 Go 开发 MCP 并部署到服务器，核心优势是编译成单个二进制文件，零运行时依赖，部署极其简洁。

第一步：编写 Go 版 MCP 服务（HTTP 模式）#

创建项目：

mkdir mcp-weather && cd mcp-weather
go mod init mcp-weather
go get github.com/mark3labs/mcp-go

编写 main.go：

package main

import (
	"context"
	"fmt"
	"log"
	"net/http"
	"os"

	"github.com/mark3labs/mcp-go/mcp"
	"github.com/mark3labs/mcp-go/server"
)

func main() {
	// 创建 MCP Server
	s := server.NewMCPServer("weather-forecast", "1.0.0")

	// 注册工具：查询未来3天天气
	weatherTool := mcp.NewTool("get_3day_forecast",
		mcp.WithDescription("查询指定城市未来3天的天气预报"),
		mcp.WithString("location",
			mcp.Required(),
			mcp.Description("城市名称，如 北京、Shanghai"),
		),
	)
	s.AddTool(weatherTool, handleWeather)

	// 以 HTTP/SSE 模式启动，监听 8080 端口
	addr := ":8080"
	if p := os.Getenv("PORT"); p != "" {
		addr = ":" + p
	}

	log.Printf("MCP Server 启动，监听 %s", addr)

	httpServer := &http.Server{Addr: addr}
	if err := server.ServeSSE(s, httpServer); err != nil {
		log.Fatalf("服务启动失败: %v", err)
	}
}
